cmake_minimum_required(VERSION 3.21)
project(AeloBrowser LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0")
set(PROJECT_ARCH "arm64")
set(USE_SANDBOX ON CACHE BOOL "Keep Chromium renderer sandbox enabled" FORCE)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
add_executable(aelo src/Browser.mm src/Workspace.mm)
target_include_directories(aelo PRIVATE "${CEF_ROOT}")
target_compile_options(aelo PRIVATE -fobjc-arc -fno-modules)
target_compile_definitions(aelo PRIVATE CEF_USE_SANDBOX)
target_link_libraries(aelo PRIVATE libcef_dll_wrapper "-framework Cocoa" "-framework QuartzCore")
add_executable(aelo_helper src/Helper.cc)
target_include_directories(aelo_helper PRIVATE "${CEF_ROOT}")
target_compile_definitions(aelo_helper PRIVATE CEF_USE_SANDBOX)
target_link_libraries(aelo_helper PRIVATE libcef_dll_wrapper "-framework Cocoa")
add_executable(workspace_tests tests/WorkspaceTests.mm src/Workspace.mm)
target_compile_options(workspace_tests PRIVATE -fobjc-arc -fno-modules)
target_link_libraries(workspace_tests PRIVATE "-framework Foundation")
enable_testing()
add_test(NAME workspace COMMAND workspace_tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME build_tools COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests -p "test_*.py")
set_tests_properties(build_tools PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
